const express = require('express');
const app = express();
const http = require('http').createServer(app);
const io = require('socket.io')(http);
const mongoose = require('mongoose');
require('dotenv').config(); // This line unlocks your .env file
const dns = require("dns");
dns.setServers(["1.1.1.1","8.8.8.8"]);
app.use(express.static('public'));
app.use(express.json());
// Define what a "Bus" looks like in the database
const busSchema = new mongoose.Schema({
    busName: String,
    lat: Number,
    lng: Number,
    lastUpdated: { type: Date, default: Date.now }
});

const Bus = mongoose.model('Bus', busSchema);
// THE MAGIC: Connect to MongoDB using your secret link
mongoose.connect(process.env.MONGO_URI)
.then(() => console.log('Successfully connected to MongoDB Database!'))
.catch((err) => console.log('Database connection error: ', err));

io.on('connection', (socket) => {
    console.log('A user opened the map dashboard!');

    let currentLat = 16.5062;
    let currentLng = 80.6480;
    // This route will receive GPS data from your Flutter app
    app.post('/update-location', async (req, res) => {
    const { busName, lat, lng } = req.body;
    
    // Update the database with the real coordinates
    await Bus.findOneAndUpdate(
        { busName: busName },
        { lat, lng, lastUpdated: new Date() },
        { upsert: true }
    );
    
    // Broadcast the update to all connected maps
    io.emit('bus-moved', { busName, lat, lng });
    
    res.sendStatus(200); // Tell the phone "We received it!"
    });
    /*const drivingSimulator = setInterval(async () => {
